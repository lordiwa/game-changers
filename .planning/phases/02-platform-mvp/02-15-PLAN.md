      #!/usr/bin/env bash
      # Idempotent provisioner for Cloud Tasks queues.
      set -euo pipefail
      PROJECT="${GCP_PROJECT_ID:-$(gcloud config get-value project)}"
      LOCATION="southamerica-east1"

      create_queue() {
        local name="$1" max_dispatches="$2" max_attempts="$3"
        gcloud tasks queues describe "$name" --location="$LOCATION" --project="$PROJECT" >/dev/null 2>&1 || \
        gcloud tasks queues create "$name" \
          --location="$LOCATION" \
          --project="$PROJECT" \
          --max-dispatches-per-second="$max_dispatches" \
          --max-attempts="$max_attempts" \
          --max-retry-duration="86400s"
        # Re-apply rate limits idempotently (update is no-op if already matching):
        gcloud tasks queues update "$name" \
          --location="$LOCATION" \
          --project="$PROJECT" \
          --max-dispatches-per-second="$max_dispatches" \
          --max-attempts="$max_attempts"
      }

      create_queue "erasure-hard-delete" 10 3
