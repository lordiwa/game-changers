       ```typescript
       import * as crypto from 'crypto';
       export async function callFunction<T>(endpoint: string, payload: any): Promise<{ ok: true; data: T } | { ok: false; error: string }> {
         const ts = Math.floor(Date.now() / 1000);
         const body = JSON.stringify({ botTimestamp: ts, payload });
         const sig = crypto.createHmac('sha256', process.env.BOT_TO_FUNCTION_HMAC!)
           .update(body).digest('hex');
         const res = await fetch(endpoint, {
           method: 'POST',
           headers: {
             'Content-Type': 'application/json',
             'Authorization': `HMAC-SHA256 t=${ts} sig=${sig}`,
           },
           body,
         });
         if (!res.ok) return { ok: false, error: `HTTP ${res.status}` };
         return res.json();
       }
       ```

        ```typescript
        export function withBotAuth(handler: (req, res, payload) => Promise<any>) {
          return async (req, res) => {
            const auth = req.header('Authorization') ?? '';
            const m = /^HMAC-SHA256 t=(\d+) sig=([a-f0-9]+)$/.exec(auth);
            if (!m) return res.status(401).end();
            const [, ts, sig] = m;
            const now = Math.floor(Date.now() / 1000);
            if (Math.abs(now - parseInt(ts, 10)) > 300) return res.status(401).end();
            const raw = JSON.stringify(req.body);
            if (!verifyHmacSha256(raw, sig, process.env.BOT_TO_FUNCTION_HMAC!)) return res.status(401).end();
            // IP allow-list
            const allowedIps = (process.env.BOT_STATIC_IPS ?? '').split(',');
            if (!allowedIps.includes(req.ip)) return res.status(403).end();
            return handler(req, res, req.body.payload);
          };
        }
        ```
